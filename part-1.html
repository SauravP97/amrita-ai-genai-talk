Creator Mode

Building the engine

import os
from google import genai

# 1. Initialize the Engine
client = genai.Client(api_key=os.environ.get("API_KEY"))

# 2. Build the System Architecture
response = client.models.generate_content(
    model="gemini-2.5-flash",
    contents="Write a python script...",
    config={
        "system_instruction": "You are an expert coder.",
        "temperature": 0.2
    }
)

print(response.text)

generate_image.py

from google import genai

client = genai.Client()

# Request an image generation
result = client.models.generate_images(
    model="imagen-3.0-generate-002",
    prompt="A futuristic cyberpunk city with flying cars and neon lights, highly detailed 3d render",
    config={"number_of_images": 1, "aspect_ratio": "1:1"}
)

# Get the generated image
print("Image Generated successfully!")
image_bytes = result.generated_images[0].image.image_bytes
