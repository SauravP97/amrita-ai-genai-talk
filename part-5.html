import os from google import genai # 1. Connect to the AI Engine client = genai.Client(api_key=os.environ.get("GEMINI_API_KEY")) # 2. The Unstructured Data messy_text = """ Hey class, the Midterm is set for Oct 15th at 10 AM. Assignments are due a few days earlier on the 12th. Also, final project presentations are scheduled for Dec 8th right before winter break! """ # 3. The Extraction Process (Magic) response = client.models.generate_content( model="gemini-2.5-flash", contents=messy_text, config={ "system_instruction": "Extract events and dates. Output ONLY a valid JSON array. No markdown.", "response_mime_type": "application/json" } ) print(response.text)
